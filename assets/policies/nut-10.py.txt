allp = []
for b in [[335, 160, 400, 245], [395, 160, 460, 245]]:
    s = observe()
    r = inspect_region(bbox_xyxy=b, sequence=s["state"]["perception"]["sequence"])
    for q in r["result"]["perception"]["points"]:
        if q[2] > 0.849:
            allp = allp + [q]
mx = 0
my = 0
n = 0
for q in allp:
    mx = mx + q[0]
    my = my + q[1]
    n = n + 1
mx = mx / n
my = my / n
print("n", n, "mean", round(mx, 4), round(my, 4))
c = 1.0
sn = 0.0
c1 = 0.9961947
s1 = 0.0871557
for k in range(36):
    pmin = 9
    pmax = -9
    qmin = 9
    qmax = -9
    for q in allp:
        p = (q[0] - mx) * c + (q[1] - my) * sn
        w = (q[1] - my) * c - (q[0] - mx) * sn
        pmin = min(pmin, p)
        pmax = max(pmax, p)
        qmin = min(qmin, w)
        qmax = max(qmax, w)
    print(k * 5, round(pmax - pmin, 4), round(qmax - qmin, 4), round(pmin, 4), round(pmax, 4), round(qmin, 4), round(qmax, 4))
    nc = c * c1 - sn * s1
    sn = sn * c1 + c * s1
    c = nc
