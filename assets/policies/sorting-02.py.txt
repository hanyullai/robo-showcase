boxes = {"can": [60, 232, 98, 278], "milk": [116, 212, 152, 276], "cereal": [168, 198, 224, 262], "bread": [160, 262, 188, 292]}
for name in ["can", "milk", "cereal", "bread"]:
    s = observe()
    r = inspect_region(bbox_xyxy=boxes[name], sequence=s["sequence"])
    p = r["result"]["perception"]
    if p["ok"]:
        g = resting_geometry(p["points"], support_z=0.82)
        pts = p["points"]
        x0 = 9
        x1 = -9
        y0 = 9
        y1 = -9
        for q in pts:
            x0 = min(x0, q[0])
            x1 = max(x1, q[0])
            y0 = min(y0, q[1])
            y1 = max(y1, q[1])
        print(name, g["ok"], g["count"], g["center"], g["z_top"], g["height"], g["base_z"], g["grasp"], "x", round(x0,3), round(x1,3), "y", round(y0,3), round(y1,3))
    else:
        print(name, "not ok", p)
