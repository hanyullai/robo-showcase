s = observe()
r = inspect_region(bbox_xyxy=[166, 198, 224, 262], sequence=s["sequence"])
p = r["result"]["perception"]
print("ok", p["ok"], p["count"])
tx = 0.0
ty = 0.0
n = 0
zt = 0.0
for q in p["points"]:
    zt = max(zt, q[2])
for q in p["points"]:
    if q[2] > zt - 0.01:
        tx = tx + q[0]
        ty = ty + q[1]
        n = n + 1
tx = tx / n
ty = ty / n
print("top", tx, ty, zt, n)
gx = tx + 0.008
gy = ty
yaw = 0.33
tol = [0.008, 0.008, 0.008]
gripper_open()
r = move_to(target_xyz=[gx, gy, 1.10], target_yaw_rad=yaw, tolerance_xyz=tol, tolerance_orientation_rad=0.08, max_steps=300, timeout_seconds=90, settle_steps=10)
print("above", r["result"]["motion"]["reached"], r["result"]["motion"]["error_max_m"])
r = move_to(target_xyz=[gx, gy, zt - 0.045], target_yaw_rad=yaw, tolerance_xyz=tol, tolerance_orientation_rad=0.08, max_steps=200, timeout_seconds=60, settle_steps=10)
print("down", r["result"]["motion"]["reached"], r["result"]["motion"]["position_xyz"])
gripper_close()
s = observe()
gp = s["state"]["gripper_positions"]
print("grip", gp)
r = move_to(target_xyz=[gx, gy, 1.12], tolerance_xyz=[0.015, 0.015, 0.015], max_steps=200, timeout_seconds=60, settle_steps=5)
print("lift", r["result"]["motion"]["reached"])
s = observe()
gp = s["state"]["gripper_positions"]
print("after lift", gp, s["state"]["end_effector_position"])
if gp[0] - gp[1] < 0.01:
    print("ABORT empty gripper")
    stop()
r = move_to(target_xyz=[0.0025, 0.4025, 1.12], tolerance_xyz=[0.015, 0.015, 0.015], max_steps=300, timeout_seconds=90, settle_steps=5)
print("carry", r["result"]["motion"]["reached"], r["result"]["motion"]["position_xyz"])
r = move_to(target_xyz=[0.0025, 0.4025, 0.98], tolerance_xyz=[0.015, 0.015, 0.015], max_steps=200, timeout_seconds=60, settle_steps=5)
print("lower", r["result"]["motion"]["reached"], r["result"]["motion"]["position_xyz"])
gripper_open()
r = move_to(target_xyz=[0.0025, 0.4025, 1.12], tolerance_xyz=[0.02, 0.02, 0.02], max_steps=200, timeout_seconds=60, settle_steps=20)
print("retreat", r["result"]["motion"]["reached"], r["result"]["motion"]["cycles"])
