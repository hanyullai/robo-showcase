s = observe()
seq = s["state"]["perception"]["sequence"]
boxes = [[345, 135, 435, 208], [428, 162, 462, 198], [365, 150, 415, 185]]
for b in boxes:
    r = inspect_region(bbox_xyxy=b, sequence=seq)
    p = r["result"]["perception"]
    pts = p["points"]
    n = len(pts)
    if n == 0:
        print(b, "none")
        continue
    mn = [9, 9, 9]
    mx = [-9, -9, -9]
    sm = [0, 0, 0]
    for q in pts:
        for i in range(3):
            mn[i] = min(mn[i], q[i])
            mx[i] = max(mx[i], q[i])
            sm[i] = sm[i] + q[i]
    print(b, p["ok"], n, "min", round(mn[0], 4), round(mn[1], 4), round(mn[2], 4), "max", round(mx[0], 4), round(mx[1], 4), round(mx[2], 4), "mean", round(sm[0] / n, 4), round(sm[1] / n, 4), round(sm[2] / n, 4))
    tx0 = 9
    tx1 = -9
    ty0 = 9
    ty1 = -9
    tn = 0
    for q in pts:
        if q[2] > mx[2] - 0.006:
            tn = tn + 1
            tx0 = min(tx0, q[0])
            tx1 = max(tx1, q[0])
            ty0 = min(ty0, q[1])
            ty1 = max(ty1, q[1])
    print(" top n", tn, "x", round(tx0, 4), round(tx1, 4), "y", round(ty0, 4), round(ty1, 4))
