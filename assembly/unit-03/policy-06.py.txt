boxes = [[230, 250, 270, 290], [368, 365, 402, 390]]
for b in boxes:
    s = observe()
    seq = s["state"]["perception"]["sequence"]
    r = inspect_region(bbox_xyxy=b, sequence=seq)
    p = r["result"]["perception"]
    if p["ok"]:
        zmax = 0
        zmin = 9
        xmin = 9
        xmax = -9
        ymin = 9
        ymax = -9
        for q in p["points"]:
            zmax = max(zmax, q[2])
            zmin = min(zmin, q[2])
            xmin = min(xmin, q[0])
            xmax = max(xmax, q[0])
            ymin = min(ymin, q[1])
            ymax = max(ymax, q[1])
        print(b, p["count"], p["centroid"], "x", xmin, xmax, "y", ymin, ymax, "z", zmin, zmax)
    else:
        print(b, "not ok", p["count"])
